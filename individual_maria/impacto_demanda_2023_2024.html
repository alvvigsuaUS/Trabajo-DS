resolver_path <- function(file_name) {
  candidates <- c(
    file.path("data", file_name),
    file.path("..", "data", file_name),
    file.path("Trabajo-DS", "data", file_name)
  )

  idx <- which(file.exists(candidates))[1]
  if (is.na(idx)) {
    stop(
      paste0(
        "No se encontró el archivo '", file_name, "'. Revisa el directorio de trabajo (getwd()) y la carpeta data/. "
      )
    )
  }

  candidates[idx]
}

# Dataset principal para análisis (incluye rama para facet)
base <- ab %>%
  inner_join(
    dem_ramas %>% select(`Rama de enseñanza`, Titulacion, Hombre, Mujer, Total),
    by = "Titulacion"
  )

# Uso explícito del dataset de centros para contraste de consistencia
dem_centros_resumen <- dem_centros %>%
  group_by(Titulacion) %>%
  summarise(
    Hombre_centros = sum(Hombre, na.rm = TRUE),
    Mujer_centros = sum(Mujer, na.rm = TRUE),
    Total_centros = sum(Total, na.rm = TRUE),
    .groups = "drop"
  )

consistencia_demanda <- base %>%
  left_join(dem_centros_resumen, by = "Titulacion") %>%
  mutate(
    diff_total_ramas_vs_centros = Total - Total_centros
  )

# Clasificación para scatter de cuadrantes (mediana abandono vs mediana demanda)
med_ab <- median(base$Tasa_Abandono, na.rm = TRUE)
med_dem <- median(base$Total, na.rm = TRUE)

base <- base %>%
  mutate(
    categoria = case_when(
      Tasa_Abandono <= med_ab & Total >= med_dem ~ "éxito",
      Tasa_Abandono > med_ab & Total >= med_dem ~ "vocacionales",
      Tasa_Abandono <= med_ab & Total < med_dem ~ "nicho",
      TRUE ~ "en crisis"
    )
  )

6) Respuesta efectiva a la pregunta

clasificar_efecto <- function(cor_val) {
  if (is.na(cor_val)) return("sin evidencia")
  if (abs(cor_val) < 0.20) return("efecto débil")
  if (abs(cor_val) < 0.40) return("efecto moderado")
  return("efecto fuerte")
}

c_ab_h <- cor(base$Tasa_Abandono, base$Hombre, use = "complete.obs")
c_ab_m <- cor(base$Tasa_Abandono, base$Mujer, use = "complete.obs")
c_re_h <- cor(base$Tasa_Rendimiento, base$Hombre, use = "complete.obs")
c_re_m <- cor(base$Tasa_Rendimiento, base$Mujer, use = "complete.obs")

q1_ab <- quantile(base$Tasa_Abandono, 0.25, na.rm = TRUE)
q1_h <- quantile(base$Hombre, 0.25, na.rm = TRUE)
q1_m <- quantile(base$Mujer, 0.25, na.rm = TRUE)

brecha_h <- base %>%
